\subsubsection{User Classes}

\noindent\textbf{User (Abstract)}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của người dùng trong hệ thống.
		\item \texttt{name: String}: Họ và tên đầy đủ của người dùng.
		\item \texttt{email: String}: Địa chỉ email được dùng để xác thực và liên lạc.
		\item \texttt{role: Role}: Vai trò của người dùng trong hệ thống, xác định quyền truy cập và chính sách áp dụng.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{login(): boolean}: Thực hiện đăng nhập vào hệ thống. Phương thức được gọi khi người dùng cung cấp thông tin xác thực; kết quả trả về \texttt{true} nếu đăng nhập thành công, ngược lại trả về \texttt{false}.
		\item \texttt{logout(): void}: Hủy phiên làm việc hiện tại và xóa trạng thái đăng nhập của người dùng. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{getRole(): Role}: Trả về vai trò hiện tại của người dùng, phục vụ cho các thao tác kiểm tra quyền và phân luồng xử lý theo nhóm đối tượng.
	\end{itemize}
\end{itemize}

\noindent\textbf{Student} (kế thừa từ \texttt{User})
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{studentId: String}: Mã số sinh viên dùng để định danh trong hệ thống quản lý học vụ và đối chiếu với \mbox{HCMUT\_DATACORE}.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{viewParkingHistory(): List<ParkingSession>}: Truy xuất danh sách các phiên gửi xe đã thực hiện của sinh viên. Kết quả trả về là danh sách \texttt{ParkingSession} được sắp xếp theo thời gian, phục vụ nhu cầu tra cứu lịch sử cá nhân.
	\end{itemize}
\end{itemize}

\noindent\textbf{Lecturer} (kế thừa từ \texttt{User})
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{staffCode: String}: Mã cán bộ của giảng viên, dùng để xác thực và tra cứu thông tin từ \mbox{HCMUT\_DATACORE}.
		\item \texttt{department: String}: Khoa hoặc bộ môn mà giảng viên trực thuộc.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{viewParkingHistory(): List<ParkingSession>}: Truy xuất lịch sử các phiên gửi xe của giảng viên. Kết quả trả về là danh sách \texttt{ParkingSession}, tương tự chức năng của \texttt{Student} nhưng áp dụng cho nhóm người dùng giảng viên.
		\item \texttt{getParkingBenefit(): PricingPolicy}: Lấy chính sách giá ưu đãi đang áp dụng cho giảng viên. Phương thức trả về đối tượng \texttt{PricingPolicy} tương ứng với vai trò \texttt{LECTURER}, phản ánh các ưu đãi như số phút miễn phí hoặc mức giảm giá được hưởng.
	\end{itemize}
\end{itemize}

\noindent\textbf{Staff} (kế thừa từ \texttt{User})
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{employeeId: String}: Mã nhân viên dùng để xác định danh tính và phân quyền vận hành trong hệ thống.
		\item \texttt{shift: String}: Ca làm việc hiện tại của nhân viên, hỗ trợ quản lý lịch trực và phân công nhiệm vụ.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{reportIncident(desc: String): Incident}: Ghi nhận sự cố phát sinh trong ca trực. Nhân viên cung cấp mô tả sự cố; hệ thống tạo đối tượng \texttt{Incident} với thông tin ban đầu và trả về sự cố vừa được tạo.
		\item \texttt{manualOverride(sessionId: String): void}: Thực hiện can thiệp thủ công vào một phiên gửi xe khi xảy ra sự cố hoặc trường hợp ngoại lệ cần xử lý trực tiếp. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{confirmPayment(sessionId: String): void}: Xác nhận thanh toán thủ công cho một phiên gửi xe, thường dùng khi hệ thống tự động gặp lỗi hoặc khách thanh toán tiền mặt tại quầy. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\noindent\textbf{Admin} (kế thừa từ \texttt{User})
\begin{itemize}
	\item \textbf{Attributes:} Không có thuộc tính riêng ngoài các thuộc tính kế thừa từ \texttt{User}.
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{configurePricing(): void}: Truy cập giao diện cấu hình chính sách giá, cho phép quản trị viên thiết lập hoặc thay đổi giá theo giờ, số phút miễn phí và mức giảm giá cho từng nhóm người dùng. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{viewReports(): Report}: Truy xuất và hiển thị báo cáo thống kê hệ thống theo loại và khoảng thời gian được chọn. Kết quả trả về là đối tượng \texttt{Report} chứa nội dung báo cáo tương ứng.
		\item \texttt{manageUsers(): void}: Thực hiện các thao tác quản lý người dùng như tra cứu, đồng bộ danh sách và kiểm tra vai trò từ \mbox{HCMUT\_DATACORE}. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\subsubsection{Entity Classes}

\noindent\textbf{Vehicle}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{licensePlate: String}: Biển số xe, dùng để định danh phương tiện trong hệ thống.
		\item \texttt{type: String}: Loại phương tiện (xe máy, ô tô, xe đạp,...), phục vụ phân loại và áp dụng chính sách phù hợp.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{validate(): boolean}: Kiểm tra tính hợp lệ của thông tin phương tiện, bao gồm định dạng biển số và loại xe. Phương thức trả về \texttt{true} nếu dữ liệu hợp lệ, ngược lại trả về \texttt{false}.
	\end{itemize}
\end{itemize}

\noindent\textbf{ParkingSession}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của phiên gửi xe.
		\item \texttt{userId: String}: Mã người dùng liên kết với phiên gửi xe; có thể để trống nếu là khách vãng lai chưa xác thực.
		\item \texttt{vehicleId: String}: Mã phương tiện được ghi nhận trong phiên gửi xe.
		\item \texttt{spotId: String}: Mã chỗ đỗ được phân công cho phiên gửi xe.
		\item \texttt{cardId: String}: Mã thẻ (thẻ nội bộ hoặc thẻ vãng lai) được dùng để mở phiên.
		\item \texttt{checkInImage: String}: Đường dẫn hoặc mã tham chiếu đến ảnh chụp phương tiện tại thời điểm vào bãi.
		\item \texttt{checkInTime: DateTime}: Thời điểm xe vào bãi, dùng để tính thời lượng gửi xe.
		\item \texttt{checkOutTime: DateTime}: Thời điểm xe ra khỏi bãi; được ghi nhận khi phiên kết thúc.
		\item \texttt{status: SessionStatus}: Trạng thái hiện tại của phiên gửi xe, có thể là \texttt{ACTIVE}, \texttt{COMPLETED} hoặc \texttt{DISPUTED}.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{startSession(): void}: Khởi tạo và kích hoạt phiên gửi xe, ghi nhận thời gian vào bãi và chuyển trạng thái phiên sang \texttt{ACTIVE}. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{endSession(): void}: Kết thúc phiên gửi xe, ghi nhận thời gian ra và chuyển trạng thái phiên sang \texttt{COMPLETED}. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{calculateDuration(): double}: Tính thời lượng gửi xe dựa trên hiệu số giữa \texttt{checkOutTime} và \texttt{checkInTime}. Kết quả trả về là số giờ dạng \texttt{double}, phục vụ bước tính phí.
	\end{itemize}
\end{itemize}

\noindent\textbf{ParkingSpot}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của chỗ đỗ xe.
		\item \texttt{zoneId: String}: Mã khu vực mà chỗ đỗ trực thuộc.
		\item \texttt{status: SpotStatus}: Trạng thái hiện tại của chỗ đỗ, có thể là \texttt{AVAILABLE}, \texttt{OCCUPIED}, \texttt{RESERVED} hoặc \texttt{ERROR}.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{occupy(): void}: Đánh dấu chỗ đỗ là đang được sử dụng bằng cách chuyển trạng thái sang \texttt{OCCUPIED}. Phương thức được gọi khi một phiên gửi xe được gán vào chỗ đỗ này. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{release(): void}: Giải phóng chỗ đỗ sau khi phiên gửi xe kết thúc, chuyển trạng thái về \texttt{AVAILABLE} để sẵn sàng tiếp nhận xe tiếp theo. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\noindent\textbf{ParkingZone}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của khu vực bãi xe.
		\item \texttt{name: String}: Tên khu vực, dùng để hiển thị trên giao diện và bảng chỉ dẫn.
		\item \texttt{totalSpots: int}: Tổng số chỗ đỗ trong khu vực.
		\item \texttt{availableSpots: int}: Số chỗ đỗ còn trống tại thời điểm hiện tại.
		\item \texttt{status: ZoneStatus}: Trạng thái tổng thể của khu vực, có thể là \texttt{AVAILABLE}, \texttt{NEARLY\_FULL} hoặc \texttt{FULL}.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{getAvailableSpots(): List<ParkingSpot>}: Trả về danh sách các chỗ đỗ còn trống trong khu vực, phục vụ bước phân công chỗ đỗ khi tạo phiên gửi xe mới.
		\item \texttt{updateStatus(): void}: Cập nhật trạng thái khu vực dựa trên số chỗ trống hiện tại so với tổng số chỗ. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\noindent\textbf{VisitorCard}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{cardId: String}: Mã định danh duy nhất của thẻ vãng lai.
		\item \texttt{status: CardStatus}: Trạng thái hiện tại của thẻ, có thể là \texttt{AVAILABLE}, \texttt{IN\_USE} hoặc \texttt{LOCKED}.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{assignToSession(session: ParkingSession): void}: Liên kết thẻ vãng lai với một phiên gửi xe cụ thể và chuyển trạng thái thẻ sang \texttt{IN\_USE}. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{reset(): void}: Đặt lại trạng thái thẻ về \texttt{AVAILABLE} sau khi phiên gửi xe kết thúc và thẻ được thu hồi. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{lock(): void}: Khóa thẻ khi phát hiện sự cố hoặc thẻ bị giữ quá thời gian cho phép, chuyển trạng thái sang \texttt{LOCKED}. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\noindent\textbf{Payment}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của giao dịch thanh toán.
		\item \texttt{sessionId: String}: Mã phiên gửi xe mà giao dịch này liên kết đến.
		\item \texttt{amount: double}: Số tiền cần thanh toán, được tính dựa trên thời lượng gửi xe và chính sách giá áp dụng.
		\item \texttt{status: PaymentStatus}: Trạng thái giao dịch, có thể là \texttt{PENDING}, \texttt{SUCCESS}, \texttt{FAILED} hoặc \texttt{CANCELLED}.
		\item \texttt{method: PaymentMethod}: Phương thức thanh toán được sử dụng, có thể là \texttt{CASH}, \texttt{QR} hoặc \texttt{PREPAID}.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{process(): boolean}: Thực hiện xử lý giao dịch thanh toán theo phương thức đã chọn. Phương thức trả về \texttt{true} nếu giao dịch thành công, ngược lại trả về \texttt{false}.
		\item \texttt{validate(): boolean}: Kiểm tra tính hợp lệ của thông tin giao dịch trước khi xử lý, bao gồm mã phiên, số tiền và phương thức thanh toán. Phương thức trả về \texttt{true} nếu dữ liệu hợp lệ, ngược lại trả về \texttt{false}.
	\end{itemize}
\end{itemize}

\noindent\textbf{PrepaidPackage}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của gói trả trước.
		\item \texttt{name: String}: Tên gói trả trước, dùng để hiển thị và phân biệt giữa các gói.
		\item \texttt{durationDays: int}: Số ngày hiệu lực của gói kể từ ngày kích hoạt.
		\item \texttt{price: double}: Giá trị của gói trả trước tính bằng đồng tiền hệ thống.
		\item \texttt{validFrom: DateTime}: Thời điểm bắt đầu hiệu lực của gói.
		\item \texttt{validTo: DateTime}: Thời điểm hết hiệu lực của gói.
		\item \texttt{userId: String}: Mã người dùng sở hữu gói trả trước này.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{isActive(): boolean}: Kiểm tra xem gói trả trước có đang trong thời hạn hiệu lực hay không, dựa trên thời gian hiện tại so với \texttt{validFrom} và \texttt{validTo}. Phương thức trả về \texttt{true} nếu gói còn hiệu lực.
		\item \texttt{extend(pkg: PrepaidPackage): void}: Gia hạn gói trả trước bằng cách cộng thêm thời gian từ gói mới vào thời điểm hết hạn hiện tại. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\noindent\textbf{PricingPolicy}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của chính sách giá.
		\item \texttt{targetGroup: Role}: Nhóm người dùng được áp dụng chính sách này, ví dụ \texttt{STUDENT}, \texttt{LECTURER}, \texttt{STAFF} hoặc \texttt{VISITOR}.
		\item \texttt{pricePerHour: double}: Mức giá tính theo giờ áp dụng cho nhóm người dùng tương ứng.
		\item \texttt{freeMinutes: int}: Số phút miễn phí được hưởng trước khi bắt đầu tính phí.
		\item \texttt{discountPercent: double}: Phần trăm giảm giá áp dụng cho nhóm đối tượng được ưu đãi.
		\item \texttt{effectiveFrom: DateTime}: Thời điểm chính sách giá bắt đầu có hiệu lực.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{calculateFee(duration: double): double}: Tính phí gửi xe dựa trên thời lượng đầu vào, áp dụng số phút miễn phí, giá theo giờ và phần trăm giảm giá theo chính sách hiện hành. Kết quả trả về là số tiền cần thanh toán dạng \texttt{double}.
	\end{itemize}
\end{itemize}

\noindent\textbf{Report}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của báo cáo.
		\item \texttt{type: ReportType}: Loại báo cáo, có thể là \texttt{REVENUE}, \texttt{USAGE} hoặc \texttt{FREQUENCY}.
		\item \texttt{dateRange: DateRange}: Khoảng thời gian mà báo cáo bao phủ.
		\item \texttt{zone: String}: Khu vực bãi xe được lọc trong báo cáo; để trống nếu báo cáo bao gồm toàn bộ hệ thống.
		\item \texttt{format: String}: Định dạng xuất báo cáo, ví dụ PDF hoặc Excel.
		\item \texttt{generatedBy: String}: Mã người dùng đã tạo báo cáo.
		\item \texttt{createdAt: DateTime}: Thời điểm báo cáo được tạo.
		\item \texttt{content: String}: Nội dung báo cáo ở dạng văn bản hoặc dữ liệu đã được tổng hợp.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{generate(): void}: Tổng hợp dữ liệu từ hệ thống theo loại báo cáo và khoảng thời gian, điền nội dung vào thuộc tính \texttt{content}. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{export(format: String): File}: Xuất báo cáo thành tệp theo định dạng yêu cầu. Kết quả trả về là đối tượng \texttt{File} để người dùng tải xuống hoặc lưu trữ.
	\end{itemize}
\end{itemize}

\noindent\textbf{Incident}
\begin{itemize}
	\item \textbf{Attributes:}
	\begin{itemize}
		\item \texttt{id: String}: Mã định danh duy nhất của sự cố.
		\item \texttt{type: IncidentType}: Loại sự cố, có thể là \texttt{HARDWARE} hoặc \texttt{ON\_SITE}.
		\item \texttt{description: String}: Mô tả chi tiết nội dung sự cố do nhân viên hoặc hệ thống ghi nhận.
		\item \texttt{status: IncidentStatus}: Trạng thái xử lý sự cố, có thể là \texttt{PENDING}, \texttt{IN\_PROGRESS} hoặc \texttt{RESOLVED}.
		\item \texttt{reportedBy: String}: Mã người dùng đã báo cáo sự cố.
		\item \texttt{createdAt: DateTime}: Thời điểm sự cố được ghi nhận vào hệ thống.
	\end{itemize}
	\item \textbf{Methods:}
	\begin{itemize}
		\item \texttt{resolve(): void}: Chuyển trạng thái sự cố sang \texttt{RESOLVED} sau khi nhân viên hoặc quản trị viên xác nhận đã xử lý xong. Phương thức không trả về dữ liệu trực tiếp.
		\item \texttt{notifyAdmin(): void}: Gửi thông báo đến quản trị viên về sự cố, bao gồm loại sự cố, mô tả, thời điểm phát sinh và trạng thái hiện tại. Phương thức không trả về dữ liệu trực tiếp.
	\end{itemize}
\end{itemize}

\subsubsection{Service Classes}

\noindent\textbf{AuthService}
\begin{itemize}
    \item \texttt{loginWithSSO(token: String): User}: Xử lý đăng nhập người dùng nội bộ thông qua \mbox{HCMUT\_SSO}. Phương thức nhận \texttt{token} từ giao diện hoặc luồng xác thực SSO, gửi token đến \texttt{SSOService} để kiểm tra, sau đó lấy thông tin người dùng và đối chiếu vai trò với DataCore khi cần. Kết quả trả về là đối tượng \texttt{User} nếu xác thực thành công; nếu token không hợp lệ, hệ thống không tạo phiên đăng nhập hợp lệ.
    \item \texttt{validateUser(user: User): boolean}: Kiểm tra đối tượng \texttt{User} trước khi cho phép truy cập hệ thống hoặc thực hiện chức năng cần phân quyền. Nội dung kiểm tra gồm mã định danh, email, vai trò, trạng thái tài khoản và quyền truy cập theo vai trò. Phương thức trả về \texttt{true} nếu người dùng hợp lệ, ngược lại trả về \texttt{false}.
\end{itemize}

\noindent\textbf{ParkingService}
\begin{itemize}
    \item \texttt{createSession(cardId: String): ParkingSession}: Tạo phiên gửi xe mới khi người dùng nội bộ hoặc khách vãng lai sử dụng thẻ tại cổng vào. Phương thức kiểm tra \texttt{cardId}, xác định loại thẻ, chọn chỗ đỗ phù hợp, ghi nhận thời gian vào bãi và khởi tạo \texttt{ParkingSession} ở trạng thái đang hoạt động. Kết quả trả về là phiên gửi xe mới nếu thẻ và chỗ đỗ hợp lệ.
    \item \texttt{closeSession(cardId: String): void}: Kết thúc phiên gửi xe khi người dùng quẹt thẻ tại cổng ra. Phương thức tìm phiên đang hoạt động theo \texttt{cardId}, ghi nhận thời gian ra, cập nhật trạng thái phiên, giải phóng chỗ đỗ và chuẩn bị dữ liệu cho bước tính phí hoặc thanh toán. Phương thức không trả về dữ liệu trực tiếp.
    \item \texttt{assignSpot(): ParkingSpot}: Chọn chỗ đỗ phù hợp cho phiên gửi xe mới dựa trên danh sách chỗ còn trống, trạng thái khu vực, vai trò người dùng và chính sách ưu tiên nếu có. Sau khi chọn được chỗ, hệ thống có thể đánh dấu chỗ là đang được sử dụng hoặc được giữ tạm thời. Kết quả trả về là \texttt{ParkingSpot} phù hợp.
    \item \texttt{updateStatus(): void}: Cập nhật trạng thái vận hành của bãi xe sau sự kiện xe vào, xe ra hoặc khi có dữ liệu mới từ IoT Gateway. Phương thức tổng hợp trạng thái phiên gửi xe, chỗ đỗ và khu vực để bảo đảm thông tin hiển thị gần thời gian thực. Phương thức không trả về dữ liệu trực tiếp.
\end{itemize}

\noindent\textbf{PaymentService}
\begin{itemize}
    \item \texttt{calculateFee(sessionId: String): double}: Tính phí gửi xe cho một phiên cụ thể. Phương thức tìm \texttt{ParkingSession} theo \texttt{sessionId}, tính thời lượng gửi xe, lấy chính sách giá hiện hành từ \texttt{PricingService}, sau đó tính số tiền cần thanh toán theo thời gian gửi, nhóm người dùng và ưu đãi nếu có. Kết quả trả về là số tiền dạng \texttt{double}.
    \item \texttt{processOnSitePayment(sessionId: String): boolean}: Xử lý thanh toán trực tiếp tại bãi xe, thường dùng cho khách vãng lai hoặc giao dịch tại cổng. Phương thức tính phí, tạo thông tin thanh toán, ghi nhận phương thức thanh toán và cập nhật trạng thái sau khi giao dịch được xác nhận. Kết quả trả về \texttt{true} khi thanh toán thành công, ngược lại trả về \texttt{false}.
    \item \texttt{processPrepaidPayment(sessionId: String): boolean}: Xử lý thanh toán bằng gói trả trước hoặc hình thức thanh toán định kỳ của người dùng nội bộ. Phương thức kiểm tra phiên gửi xe, xác định người dùng, kiểm tra gói trả trước hoặc chính sách đang áp dụng, sau đó ghi nhận giao dịch tương ứng. Kết quả trả về \texttt{true} nếu giao dịch hợp lệ và được ghi nhận.
\end{itemize}

\noindent\textbf{PricingService}
\begin{itemize}
    \item \texttt{getCurrentPolicy(role: Role): PricingPolicy}: Lấy chính sách giá hiện hành cho một nhóm người dùng. Phương thức sử dụng \texttt{role} để xác định đối tượng áp dụng như sinh viên, giảng viên, nhân viên, quản trị viên hoặc khách vãng lai. Kết quả trả về là \texttt{PricingPolicy} tương ứng.
    \item \texttt{updatePolicy(policy: PricingPolicy): void}: Cập nhật hoặc thay thế chính sách giá trong hệ thống. Chức năng này được dùng bởi quản trị viên khi thay đổi giá theo giờ, số phút miễn phí, phần trăm giảm giá hoặc nhóm người dùng được áp dụng. Phương thức nhận \texttt{policy} mới hoặc đã chỉnh sửa và không trả về dữ liệu trực tiếp.
\end{itemize}

\noindent\textbf{ReportService}
\begin{itemize}
    \item \texttt{generateReport(type: ReportType, dateRange: DateRange): Report}: Tạo báo cáo thống kê theo loại báo cáo và khoảng thời gian được chọn. Phương thức truy xuất dữ liệu về phiên gửi xe, giao dịch thanh toán, trạng thái khu vực hoặc tần suất sử dụng, sau đó tổng hợp thành nội dung báo cáo. Kết quả trả về là đối tượng \texttt{Report}.
    \item \texttt{exportReport(id: String, format: String): File}: Xuất báo cáo đã tạo thành tệp theo định dạng được yêu cầu. Phương thức tìm báo cáo theo \texttt{id}, kiểm tra \texttt{format}, tạo tệp và cung cấp kết quả cho người dùng tải xuống hoặc lưu trữ. Kết quả trả về là đối tượng \texttt{File}.
\end{itemize}

\noindent\textbf{IncidentService}
\begin{itemize}
    \item \texttt{reportIncident(desc: String, type: IncidentType): Incident}: Ghi nhận sự cố phát sinh trong quá trình vận hành bãi xe, bao gồm lỗi phần cứng, thiết bị IoT, cổng ra vào, thanh toán hoặc tình huống cần nhân viên xử lý. Phương thức tạo \texttt{Incident} với mô tả, loại sự cố, trạng thái ban đầu và người báo cáo nếu có. Kết quả trả về là sự cố mới được tạo.
    \item \texttt{resolveIncident(id: String): void}: Cập nhật trạng thái sự cố sau khi nhân viên hoặc quản trị viên xử lý xong. Phương thức tìm sự cố theo \texttt{id}, kiểm tra trạng thái hiện tại và chuyển sự cố sang trạng thái đã giải quyết khi thao tác hợp lệ. Phương thức không trả về dữ liệu trực tiếp.
    \item \texttt{notifyAdmin(incident: Incident): void}: Gửi thông báo đến quản trị viên khi có sự cố cần theo dõi hoặc xử lý. Nội dung thông báo có thể gồm loại sự cố, mô tả, thời điểm phát sinh và trạng thái hiện tại. Phương thức không trả về dữ liệu trực tiếp.
\end{itemize}

\noindent\textbf{ZoneService}
\begin{itemize}
    \item \texttt{getZoneStatus(): List<ParkingZone>}: Lấy danh sách trạng thái hiện tại của các khu vực trong bãi xe. Phương thức tổng hợp số chỗ trống, tổng số chỗ, trạng thái khu vực và dữ liệu cảm biến nếu có. Kết quả trả về là danh sách \texttt{ParkingZone}.
    \item \texttt{updateZoneStatus(zoneId: String): void}: Cập nhật trạng thái của một khu vực bãi xe cụ thể. Phương thức xác định số chỗ trống hiện tại trong khu vực, so sánh với tổng số chỗ và cập nhật trạng thái khu vực thành còn chỗ, gần đầy hoặc đầy. Phương thức không trả về dữ liệu trực tiếp.
    \item \texttt{suggestZone(role: Role): ParkingZone}: Đề xuất khu vực đỗ xe phù hợp dựa trên vai trò người dùng và trạng thái hiện tại của các khu vực. Hệ thống có thể ưu tiên khu vực dành cho nhóm người dùng tương ứng hoặc khu vực thay thế khi khu ưu tiên đã đầy. Kết quả trả về là \texttt{ParkingZone} được đề xuất.
\end{itemize}

\subsubsection{Controller Classes}

\noindent\textbf{AuthController}
\begin{itemize}
    \item \texttt{login(): void}: Tiếp nhận yêu cầu đăng nhập từ giao diện người dùng, lấy token hoặc dữ liệu xác thực cần thiết và gọi \texttt{AuthService.loginWithSSO()} để xử lý. Phương thức không có tham số trong class diagram và không trả về dữ liệu trực tiếp; kết quả được thể hiện qua phản hồi đăng nhập thành công hoặc thất bại.
    \item \texttt{logout(): void}: Tiếp nhận yêu cầu đăng xuất, hủy phiên đăng nhập hiện tại và xóa trạng thái phiên nếu cần. Phương thức không có tham số và không trả về dữ liệu trực tiếp.
\end{itemize}

\noindent\textbf{ParkingController}
\begin{itemize}
    \item \texttt{checkIn(): void}: Tiếp nhận yêu cầu vào bãi xe từ cổng vào sau khi người dùng quẹt thẻ hoặc thẻ vãng lai được cấp. Controller lấy dữ liệu thẻ từ request hoặc thiết bị đọc thẻ, gọi \texttt{ParkingService.createSession()} và điều phối mở barrier nếu phiên gửi xe được tạo thành công.
    \item \texttt{checkOut(): void}: Tiếp nhận yêu cầu ra khỏi bãi xe tại cổng ra. Controller lấy mã thẻ, gọi \texttt{ParkingService.closeSession()}, sau đó điều phối bước tính phí hoặc xác nhận thanh toán trước khi mở barrier. Phương thức không trả về dữ liệu trực tiếp.
    \item \texttt{getZoneStatus(): void}: Tiếp nhận yêu cầu xem trạng thái các khu vực bãi xe từ người dùng, nhân viên hoặc bảng hiển thị. Controller gọi \texttt{ZoneService.getZoneStatus()} và cung cấp dữ liệu trạng thái cho giao diện hoặc thiết bị hiển thị.
\end{itemize}

\noindent\textbf{PaymentController}
\begin{itemize}
    \item \texttt{payOnSite(): void}: Tiếp nhận yêu cầu thanh toán trực tiếp tại bãi xe. Controller lấy mã phiên gửi xe từ request hoặc từ thẻ, gọi \texttt{PaymentService.processOnSitePayment()} và trả kết quả thanh toán cho nhân viên hoặc thiết bị tại cổng.
    \item \texttt{payPrepaid(): void}: Tiếp nhận yêu cầu thanh toán bằng gói trả trước hoặc thanh toán định kỳ. Controller lấy thông tin phiên gửi xe hoặc người dùng, gọi \texttt{PaymentService.processPrepaidPayment()} và phản hồi kết quả xử lý cho giao diện.
\end{itemize}

\noindent\textbf{AdminController}
\begin{itemize}
    \item \texttt{configurePricing(): void}: Tiếp nhận yêu cầu cấu hình chính sách giá từ quản trị viên. Controller lấy thông tin chính sách mới từ giao diện, kiểm tra quyền thao tác cơ bản và gọi \texttt{PricingService.updatePolicy()} để cập nhật chính sách.
    \item \texttt{manageUsers(): void}: Tiếp nhận yêu cầu quản lý người dùng từ quản trị viên. Controller có thể gọi \texttt{DataCoreService.getUser()} hoặc \texttt{DataCoreService.syncUserList()} để lấy và đồng bộ dữ liệu người dùng từ \mbox{HCMUT\_DATACORE}, phục vụ kiểm tra thông tin và vai trò.
    \item \texttt{monitorSystem(): void}: Tiếp nhận yêu cầu giám sát trạng thái hệ thống từ quản trị viên. Controller tổng hợp trạng thái đồng bộ DataCore, kết nối IoT Gateway, trạng thái khu vực bãi xe và các sự cố đang tồn tại để hiển thị trên giao diện quản trị.
\end{itemize}

\noindent\textbf{ReportController}
\begin{itemize}
    \item \texttt{viewReport(): void}: Tiếp nhận yêu cầu xem báo cáo từ nhân viên hoặc quản trị viên. Controller lấy loại báo cáo, khoảng thời gian và tiêu chí lọc từ giao diện, sau đó gọi \texttt{ReportService.generateReport()} để tạo và hiển thị báo cáo.
    \item \texttt{exportReport(): void}: Tiếp nhận yêu cầu xuất báo cáo từ người dùng có quyền. Controller lấy mã báo cáo và định dạng tệp, gọi \texttt{ReportService.exportReport()} và cung cấp tệp báo cáo cho người dùng.
\end{itemize}

\noindent\textbf{IncidentController}
\begin{itemize}
    \item \texttt{submitIncident(): void}: Tiếp nhận yêu cầu ghi nhận sự cố từ nhân viên hoặc giao diện vận hành. Controller lấy mô tả và loại sự cố, gọi \texttt{IncidentService.reportIncident()} và phản hồi kết quả ghi nhận cho giao diện.
    \item \texttt{resolveIncident(): void}: Tiếp nhận yêu cầu xử lý hoặc đóng sự cố. Controller lấy mã sự cố, gọi \texttt{IncidentService.resolveIncident()} và cập nhật trạng thái sự cố trên giao diện nếu thao tác hợp lệ.

Phương thức được dùng khi giao dịch phát sinh lỗi, người dùng dừng thanh toán hoặc phiên gửi xe không còn hợp lệ. Kết quả trả về \texttt{true} nếu hủy thành công, ngược lại trả về \texttt{false}.
\end{itemize}

\noindent\textbf{DataCoreService}

    \item \texttt{syncUserList(): List<User>}: Đồng bộ danh sách người dùng từ \mbox{HCMUT\_DATACORE} về hệ thống Smart Parking. Phương thức hỗ trợ cập nhật danh sách người dùng và vai trò theo nguồn dữ liệu chính thức. Kết quả trả về là danh sách \texttt{User}.

\end{itemize}

\noindent\textbf{IoTGateway}

    \item \texttt{receiveData(): SensorData}: Nhận dữ liệu cảm biến từ IoT Gateway, bao gồm trạng thái chỗ đỗ, trạng thái thiết bị, thời điểm cập nhật và dữ liệu phục vụ giám sát gần thời gian thực. Kết quả trả về là \texttt{SensorData}.
    \item \texttt{openBarrier(): void}: Gửi yêu cầu mở barrier tại cổng ra hoặc cổng vào. Phương thức thường được gọi sau khi hệ thống xác nhận người dùng được phép vào bãi hoặc ra khỏi bãi. Phương thức không trả về dữ liệu trực tiếp.
    \item \texttt{getConnectionStatus(): boolean}: Kiểm tra trạng thái kết nối giữa hệ thống trung tâm và IoT Gateway. Phương thức trả về \texttt{true} nếu gateway hoạt động bình thường, ngược lại trả về \texttt{false}.
    \item \texttt{syncOfflineData(): void}: Đồng bộ dữ liệu được lưu tạm trong gateway khi kết nối với hệ thống trung tâm bị gián đoạn. Sau khi kết nối được khôi phục, gateway gửi lại dữ liệu chưa đồng bộ như trạng thái cảm biến, sự kiện vào ra hoặc bản ghi thiết bị. Phương thức không trả về dữ liệu trực tiếp.
\end{itemize}

	\end{itemize}

\subsubsection{Enumerations}

\noindent\textbf{Role}
\begin{itemize}
	\item \texttt{STUDENT}: Sinh viên đang theo học tại trường, được áp dụng chính sách giá và quyền truy cập dành riêng cho nhóm này.
	\item \texttt{LECTURER}: Giảng viên hoặc cán bộ giảng dạy, được hưởng ưu đãi về phí gửi xe so với khách vãng lai.
	\item \texttt{STAFF}: Nhân viên vận hành bãi xe, có quyền thực hiện các thao tác trực tiếp như xác nhận thanh toán và ghi nhận sự cố.
	\item \texttt{ADMIN}: Quản trị viên hệ thống, có toàn quyền cấu hình chính sách, xem báo cáo và quản lý người dùng.
	\item \texttt{VISITOR}: Khách vãng lai không thuộc hệ thống nội bộ, sử dụng thẻ tạm thời và áp dụng mức giá tiêu chuẩn.
\end{itemize}

\noindent\textbf{SessionStatus}
\begin{itemize}
	\item \texttt{ACTIVE}: Phiên gửi xe đang diễn ra, xe đã vào bãi và chưa ra khỏi cổng.
	\item \texttt{COMPLETED}: Phiên gửi xe đã kết thúc bình thường, xe đã ra khỏi bãi và thanh toán được xử lý.
	\item \texttt{DISPUTED}: Phiên gửi xe đang có tranh chấp hoặc bất thường, cần nhân viên hoặc quản trị viên xem xét và xử lý thủ công.
\end{itemize}

\noindent\textbf{SpotStatus}
\begin{itemize}
	\item \texttt{AVAILABLE}: Chỗ đỗ đang trống và sẵn sàng tiếp nhận phương tiện mới.
	\item \texttt{OCCUPIED}: Chỗ đỗ đang có phương tiện, liên kết với một phiên gửi xe đang hoạt động.
	\item \texttt{RESERVED}: Chỗ đỗ đã được giữ tạm thời trong quá trình tạo phiên gửi xe, chưa chính thức sử dụng.
	\item \texttt{ERROR}: Chỗ đỗ gặp sự cố kỹ thuật hoặc dữ liệu không nhất quán, cần kiểm tra trước khi đưa vào sử dụng.
\end{itemize}

\noindent\textbf{ZoneStatus}
\begin{itemize}
	\item \texttt{AVAILABLE}: Khu vực còn nhiều chỗ trống, hệ thống có thể tiếp tục phân công chỗ đỗ bình thường.
	\item \texttt{NEARLY\_FULL}: Khu vực gần đầy, số chỗ trống còn lại ở mức thấp; hệ thống có thể ưu tiên đề xuất khu vực khác.
	\item \texttt{FULL}: Khu vực đã hết chỗ, không thể tiếp nhận thêm phương tiện cho đến khi có chỗ được giải phóng.
\end{itemize}

\noindent\textbf{PaymentStatus}
\begin{itemize}
	\item \texttt{PENDING}: Giao dịch thanh toán đã được tạo nhưng chưa hoàn tất xử lý.
	\item \texttt{SUCCESS}: Giao dịch thanh toán đã được xử lý thành công và xác nhận bởi hệ thống.
	\item \texttt{FAILED}: Giao dịch thanh toán thất bại do lỗi kỹ thuật, không đủ số dư hoặc hết thời gian chờ.
	\item \texttt{CANCELLED}: Giao dịch thanh toán đã bị hủy theo yêu cầu của người dùng hoặc do hệ thống phát hiện phiên không hợp lệ.
\end{itemize}

\noindent\textbf{PaymentMethod}
\begin{itemize}
	\item \texttt{CASH}: Thanh toán tiền mặt trực tiếp tại quầy hoặc cổng ra, được nhân viên xác nhận thủ công.
	\item \texttt{QR}: Thanh toán qua mã QR thông qua hệ thống \mbox{BKPay}, phù hợp cho cả người dùng nội bộ và khách vãng lai.
	\item \texttt{PREPAID}: Thanh toán bằng gói trả trước đã đăng ký, áp dụng cho người dùng nội bộ có gói còn hiệu lực.
\end{itemize}

\noindent\textbf{CardStatus}
\begin{itemize}
	\item \texttt{AVAILABLE}: Thẻ vãng lai đang sẵn sàng, chưa được gán cho phiên gửi xe nào.
	\item \texttt{IN\_USE}: Thẻ đang được liên kết với một phiên gửi xe đang hoạt động.
	\item \texttt{LOCKED}: Thẻ bị khóa do phát hiện sự cố hoặc giữ thẻ quá thời gian quy định, cần nhân viên xử lý trước khi tái sử dụng.
\end{itemize}

\noindent\textbf{IncidentType}
\begin{itemize}
	\item \texttt{HARDWARE}: Sự cố liên quan đến thiết bị vật lý như đầu đọc thẻ RFID, camera, barrier hoặc IoT Gateway.
	\item \texttt{ON\_SITE}: Sự cố phát sinh trực tiếp tại bãi xe như tranh chấp chỗ đỗ, lỗi thanh toán hoặc tình huống cần nhân viên can thiệp.
\end{itemize}

\noindent\textbf{IncidentStatus}
\begin{itemize}
	\item \texttt{PENDING}: Sự cố vừa được ghi nhận, chưa có nhân viên hoặc quản trị viên tiếp nhận xử lý.
	\item \texttt{IN\_PROGRESS}: Sự cố đang được xử lý bởi nhân viên hoặc quản trị viên có thẩm quyền.
	\item \texttt{RESOLVED}: Sự cố đã được giải quyết hoàn toàn và đóng lại trong hệ thống.
\end{itemize}

\noindent\textbf{ReportType}
\begin{itemize}
	\item \texttt{REVENUE}: Báo cáo doanh thu, tổng hợp số tiền thu được từ các giao dịch thanh toán trong khoảng thời gian được chọn.
	\item \texttt{USAGE}: Báo cáo tình hình sử dụng bãi xe, bao gồm số lượt gửi xe, tỷ lệ lấp đầy và phân bố theo khu vực.
	\item \texttt{FREQUENCY}: Báo cáo tần suất sử dụng, phân tích mức độ hoạt động của bãi xe theo khung giờ, ngày hoặc nhóm người dùng.
\end{itemize}
